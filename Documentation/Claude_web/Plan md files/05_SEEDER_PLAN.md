## 🗂️ SEEDER 2 — OTP Tokens (14 records — all pre-used)

One token per user, all marked `used: true` so they don't interfere with real OTP flow.

```typescript
// server/prisma/seeders/02-otp-tokens.seeder.ts
import bcrypt from 'bcrypt'
import { PrismaClient } from '@prisma/client'

export async function seedOtpTokens(
  prisma:  PrismaClient,
  userIds: Record<string, string>
) {
  console.log('🌱 Seeding OTP tokens...')

  for (const [email, userId] of Object.entries(userIds)) {
    const code     = '123456'                       // dummy code — already used
    const codeHash = await bcrypt.hash(code, 10)
    const expiry   = new Date(Date.now() - 60_000)  // already expired

    await prisma.otpToken.create({
      data: {
        userId,
        codeHash,
        expiresAt: expiry,
        used:      true,       // ← never valid for login
      },
    })
  }
  console.log(`  ✅ ${Object.keys(userIds).length} OTP tokens seeded (all pre-used)`)
}
```

```typescript
// server/prisma/seeders/05-availability.seeder.ts
import { PrismaClient } from '@prisma/client'
import { addDays, format } from 'date-fns'

export async function seedAvailability(
  prisma:      PrismaClient,
  roomTypeIds: string[]
) {
  console.log('🌱 Seeding availability (90 days)...')

  const today = new Date()
  today.setHours(0, 0, 0, 0)

  // Build all rows first, then bulk insert in batches
  const rows: Array<{
    roomTypeId:  string
    date:        Date
    isBlocked:   boolean
    bookingId:   null
  }> = []

  for (const roomTypeId of roomTypeIds) {
    for (let i = 0; i < 90; i++) {
      const date = addDays(today, i)
      rows.push({
        roomTypeId,
        date,
        isBlocked: false,
        bookingId: null,
      })
    }
  }

  // Batch insert in chunks of 500 to avoid hitting DB limits
  const CHUNK = 500
  let inserted = 0
  for (let i = 0; i < rows.length; i += CHUNK) {
    await prisma.availability.createMany({
      data:           rows.slice(i, i + CHUNK),
      skipDuplicates: true,
    })
    inserted += Math.min(CHUNK, rows.length - i)
  }

  console.log(`  ✅ ${inserted} availability rows inserted`)
}
```

## 🧪 Verification Queries (run in pgAdmin after seed)

```sql
-- Count all records
SELECT 'users'                AS tbl, COUNT(*) FROM users
UNION ALL SELECT 'properties',         COUNT(*) FROM properties
UNION ALL SELECT 'room_types',         COUNT(*) FROM room_types
UNION ALL SELECT 'availability',       COUNT(*) FROM availability
UNION ALL SELECT 'bookings',           COUNT(*) FROM bookings
UNION ALL SELECT 'payment_simulations',COUNT(*) FROM payment_simulations;

-- Check category-amenity match
SELECT name, category, amenities FROM properties ORDER BY category;

-- Upcoming bookings
SELECT b.confirmation_number, u.name, p.name, b.checkin, b.checkout, b.status
FROM bookings b
JOIN users u ON u.id = b.user_id
JOIN properties p ON p.id = b.property_id
WHERE b.checkin > NOW()
ORDER BY b.checkin;

-- Verify full-text search works
SELECT name, city FROM properties
WHERE search_vector @@ to_tsquery('english', 'Goa & resort');
```
